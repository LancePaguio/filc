// division
isulat(7 / 2); // expect: 3
isulat(-7 / 2); // expect: -3

// modulo
isulat(7 % 3); // expect: 1
isulat(-7 % 3); // expect: -1
isulat(7 % -3); // expect: 1

// negation
isulat(-(3)); // expect: -3
